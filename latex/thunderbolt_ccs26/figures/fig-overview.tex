%% Figure 1: "Dummy" icon-based overview — how 1 BTC moves from Alice to Bob
%% Previous version: fig-overview-v0.tex
%%
%% Colour scheme (Wong 2011 colorblind-safe palette):
%%   Alice = amber orange,  Bob = deep blue,  Committee = rose-purple,  Lock = vermilion
%%
\definecolor{cAlice}{HTML}{E69F00}   % amber orange (colorblind-safe)
\definecolor{cBob}{HTML}{0072B2}     % deep blue (colorblind-safe)
\definecolor{cComm}{HTML}{CC79A7}    % rose-purple (colorblind-safe)
\definecolor{cLedger}{HTML}{555555}   % dark gray
\definecolor{cLock}{HTML}{D55E00}     % vermilion for lock/public key (colorblind-safe)

\begin{figure*}[t]
\centering
\resizebox{0.96\textwidth}{!}{%
\begin{tikzpicture}[
  A/.style={-{Stealth[length=3.2mm, width=2.8mm]}, line width=1.3pt},
  Adash/.style={A, densely dashed, color=black!50},
  actor/.style={rounded corners=4pt, inner sep=7pt,
                minimum width=3.2cm, minimum height=1.1cm,
                font=\LARGE\bfseries, fill=white, align=center, draw=black!60},
  sbox/.style={draw=black!45, rounded corners=3pt, inner sep=7pt,
               font=\Large, fill=white, align=left, text=black!80},
  stepnum/.style={circle, draw=black!60, fill=white, inner sep=2pt,
                  font=\LARGE\bfseries, minimum size=0.7cm},
  lifeline/.style={densely dotted, black!18, line width=0.7pt},
  phasebar/.style={draw=none, fill=black!8, rounded corners=2pt,
                   minimum height=0.6cm, inner sep=6pt,
                   font=\LARGE\bfseries, text=black!50},
  arrlbl/.style={font=\Large, text=black!60},
  arrsub/.style={font=\Large\itshape, text=black!40},
]

%% ── Column positions (Alice, Committee, Bob, Ledger) ──
\def\xAlice{0}
\def\xComm{7}
\def\xBob{14}
\def\xLedger{21}

%% ── Actor headers (with icons) ──
\node[actor, fill=cAlice!10, draw=cAlice!60] (alice) at (\xAlice, 0)
  {{\color{cAlice}\faUser}\; Alice\\[-2pt]{\Large current owner}};
\node[actor, fill=cComm!10, draw=cComm!60] (comm) at (\xComm, 0)
  {{\color{cComm}\faUsers}\; Committee\\[-2pt]{\Large semi-trusted}};
\node[actor, fill=cBob!8, draw=cBob!60] (bob) at (\xBob, 0)
  {{\color{cBob}\faUser}\; Bob\\[-2pt]{\Large recipient}};
\node[actor, fill=black!6, draw=cLedger!60, minimum width=3.6cm] (ledger) at (\xLedger, 0)
  {{\color{cLedger}\faDatabase}\; Ledger\\[-2pt]{\Large append-only}};

%% ── Lifelines ──
\draw[lifeline] (\xAlice, -0.7) -- (\xAlice, -22.8);
\draw[lifeline] (\xComm,  -0.7) -- (\xComm,  -22.8);
\draw[lifeline] (\xBob,   -0.7) -- (\xBob,   -22.8);
\begin{scope}[on background layer]
\fill[black!3, rounded corners=2pt]
  (\xLedger-0.18, -0.7) rectangle (\xLedger+0.18, -22.8);
\end{scope}
\draw[lifeline, black!25] (\xLedger, -0.7) -- (\xLedger, -22.8);

%% ════════════════════════════════════════════════════════
%% PHASE: Open
%% ════════════════════════════════════════════════════════

\node[phasebar] at (-3.8, -1.5) {Open};
\node[stepnum] at (-2.3, -1.5) {1};

%% Visual equation: Alice-key + Committee-key = lock
\node[sbox, fill=cAlice!8] (oa) at (\xAlice, -2.4)
  {{\color{cAlice}\faKey}\; holds \textbf{half-key}};
\node[sbox, fill=cComm!8] (oc) at (\xComm, -2.4)
  {{\color{cComm}\faKey}\; holds \textbf{half-key}};

\draw[A, <->] (\xAlice+2.0, -3.2) --
  node[arrlbl, above] {{\color{cAlice}\faKey}\;$+$\;{\color{cComm}\faKey}\;$=$\;{\Large\color{cLock}\faLock}}
  (\xComm-2.0, -3.2);

%% UTXO box
\node[draw=black!55, rounded corners=4pt, fill=black!5, inner sep=7pt,
      font=\Large, align=center, minimum width=8cm]
  (utxo) at (3.5, -4.2)
  {{\LARGE\faBitcoin}\; 1\,BTC \quad locked by \;{\Large\color{cLock}\faLock}};
\node[arrsub, anchor=west] at (utxo.east) {\;\; on-chain tx \#1};

%% ════════════════════════════════════════════════════════
%% PHASE: Transfer
%% ════════════════════════════════════════════════════════

\draw[black!12, line width=0.5pt] (-4.3, -5.4) -- (24.0, -5.4);
\node[phasebar] at (-3.8, -6.0) {Transfer};
\node[arrsub, anchor=west] at (-1.5, -6.0) {entirely off-chain};
\node[font=\Large\bfseries, text=black!70, anchor=west] at (-4.3, -7.6)
  {\raisebox{2pt}{\tikz\draw[densely dashed, black!50, line width=1.3pt, -{Stealth[length=2.6mm]}] (0,0)--(1.4,0);}\;\; = asynchronous (Alice need not be online)};
\node[font=\Large\bfseries, text=black!70, anchor=west] at (-4.3, -8.4)
  {\raisebox{2pt}{\tikz\draw[black, line width=1.3pt, -{Stealth[length=2.6mm]}] (0,0)--(1.4,0);}\;\; = synchronous (Alice online)};

%% ── Step 2: Bob creates invoice ──
\node[sbox, fill=cBob!6, anchor=north] (s2) at (\xBob, -6.8) {%
  \begin{tabular}{@{}l@{}}
  Pick a fresh refresh value {\color{cBob}\faKey}\\[4pt]
  Seal three copies of it:\\[3pt]
  \quad {\normalsize\faEye}\; public commitment\\[3pt]
  \quad {\color{cBob}\faEnvelope}\; encrypted for self\\[3pt]
  \quad {\color{cComm}\faEnvelope}\; encrypted for Committee\\[4pt]
  Attach proof $\pi_{\mathrm{inv}}$\\[-1pt]
  {\large\itshape\color{black!45} ``all three hide the same secret''}
  \end{tabular}
};
\node[stepnum] at ([shift={(-6pt,6pt)}]s2.north west) {2};

%% ── Step 3: Alice processes ──
\node[sbox, fill=cAlice!8, anchor=north] (s3) at (\xAlice, -11.0) {%
  \begin{tabular}{@{}l@{}}
  Check proof $\pi_{\mathrm{inv}}$\\[4pt]
  Homomorphically add:\\[3pt]
  \quad {\color{cAlice}\faKey}\;$+$\;{\color{cBob}\faEnvelope}%
    \;$\to$\;{\color{black!60}\faEnvelope}\; credential\\[2pt]
  \quad {\large\itshape\color{black!45} encrypts %
    {\color{cAlice}\faKey}\,$+$\,{\color{cBob}\faKey}}\\[4pt]
  Attach proof $\pi_{\mathrm{xfer}}$\\[-1pt]
  {\large\itshape\color{black!45} ``keys correctly combined''}
  \end{tabular}
};
\node[stepnum] at ([shift={(-6pt,6pt)}]s3.north west) {3};

%% ── Step 4 boxes (4a above 4b) ──
\node[sbox, fill=cBob!6, anchor=north] (s4a) at (\xBob, -16.0) {%
  \begin{tabular}{@{}l@{}}
  {\color{black!60}\faEnvelopeOpen}\; Decrypt credential\\[3pt]
  Obtains {\color{cAlice}\faKey}\,$+$\,{\color{cBob}\faKey}
  \end{tabular}
};
\node[stepnum] at ([shift={(-6pt,6pt)}]s4a.north west) {4a};

\node[sbox, fill=cComm!6, anchor=north] (s4b) at (\xComm, -17.8) {%
  \begin{tabular}{@{}l@{}}
  {\color{cComm}\faEnvelopeOpen}\; Decrypt secret\\[3pt]
  Computes {\color{cComm}\faKey}\,$-$\,{\color{cBob}\faKey}
  \end{tabular}
};
\node[stepnum] at ([shift={(-6pt,6pt)}]s4b.north west) {4b};

%% ════════════════════════════════════════════════════════
%% ARROWS — all aligned to box edges using node anchors
%% ════════════════════════════════════════════════════════

%% Arrow 1: Bob publishes invoice → Ledger (dashed = Alice not online)
\draw[Adash] (s2.east) -- node[arrlbl, above] {publish invoice} (s2.east -| ledger);

%% Arrow 2: Alice fetches invoice ← Ledger (solid = Alice online)
\coordinate (fi) at ($(s3.north east)!0.2!(s3.south east)$);
\draw[A] (fi -| ledger) -- node[arrlbl, above] {fetch invoice} (fi);

%% Arrow 3: Alice publishes credential → Ledger (solid = Alice online)
\coordinate (pc) at ($(s3.north east)!0.8!(s3.south east)$);
\draw[A] (pc) -- node[arrlbl, above] {publish credential} (pc -| ledger);

%% Arrow 4: Bob fetches credential ← Ledger (dashed = Alice not online)
\draw[Adash] (s4a.east -| ledger) -- node[arrlbl, above] {fetch credential} (s4a.east);

%% Arrow 5: Committee fetches encrypted envelope ← Ledger (dashed = Alice not online)
\draw[Adash] (s4b.east -| ledger) --
  node[arrlbl, above] {fetch encrypted {\color{cComm}\faEnvelope}} (s4b.east);

\node[arrsub, anchor=north] at ($(s4b.south east)!0.5!(s4b.south east -| ledger) + (0,-0.3)$) {4a and 4b proceed independently};

%% ════════════════════════════════════════════════════════
%% PHASE: Finalize
%% ════════════════════════════════════════════════════════

\draw[black!12, line width=0.5pt] (-4.3, -19.6) -- (24.0, -19.6);
\node[phasebar] at (-3.8, -20.2) {Finalize};
\node[stepnum] at (-2.3, -20.2) {5};

%% Visual: key cancellation → unlock → BTC spent
\node[draw=black!55, rounded corners=4pt, fill=black!5, inner sep=8pt,
      font=\Large, align=center, minimum width=21cm]
  (spend) at (10.5, -21.2)
  {$\underbrace{\text{{\color{cAlice}\faKey}$\,+\,${\color{cBob}\faKey}}}_{\text{\large Bob's share}}$%
   \;$+$\;%
   $\underbrace{\text{{\color{cComm}\faKey}$\,-\,${\color{cBob}\faKey}}}_{\text{\large Committee's share}}$%
   \quad$=$\quad
   {\color{cAlice}\faKey}\;$+$\;{\color{cComm}\faKey}%
   \;$=$\;{\Large\color{cLock}\faLock}%
   \quad$\longrightarrow$\quad
   {\color{green!50!black}\faLockOpen}
   \quad$\longrightarrow$\quad
   {\LARGE\faBitcoin}\; Bob spends 1\,BTC};
\node[arrsub, anchor=west] at (spend.east) {\;\; on-chain tx \#2};

%% ── Phantom bounds ──
\path (-4.8, -23.4) (24.5, 1.0);

\end{tikzpicture}%
}
\caption{End-to-end overview of \thunderbolt: transferring 1\,BTC from Alice to Bob via the Thunderbolt Ledger. The chain sees exactly two transactions; all transfers happen off-chain.}
\label{fig:overview}
\Description{Icon-based sequence diagram of the Thunderbolt protocol lifecycle: Open (fund UTXO under a combined lock-key), Transfer (off-chain invoice-and-consume via the Thunderbolt Ledger), and Finalize (co-sign spending transaction).}
\end{figure*}
